	\begin{align*}
		\dfrac{\partial u}{\partial t}&=-\dfrac{\partial p}{\partial x}+F_0+Y_iF_i+Y_iY_jF_{ij}+f_{x},
	\end{align*}
	where
	\begin{align*}
		f_x&=f_{x,det}+f_{x,stoch},\\
		F_0&=-\bar u\dfrac{\partial\bar u}{\partial x}-\bar v\dfrac{\partial\bar u		}{\partial y}-\bar w\dfrac{\partial\bar u}{\partial z}+\nu\Delta\bar u,\\
		F_i&=-\left(\bar u\dfrac{\partial u_i}{\partial x}+u_i\dfrac{\partial\bar u}{\partial x}+\bar v\dfrac{\partial u_i}{\partial y}+v_i\dfrac{\partial\bar u}{\partial y}+\bar w\dfrac{\partial u_i}{\partial z}+w_i\dfrac{\partial\bar u}{\partial z}\right)+\nu\Delta u_i,\\
		F_{ij}&=-\left(u_i\dfrac{\partial u_j}{\partial x}+v_i\dfrac{\partial u_j}{\partial y}+w_i\dfrac{\partial u_j}{\partial z}\right)
	\end{align*}
	
	Hence,
	\begin{align*}
		\dfrac{\partial\bar u}{\partial t}&=-\dfrac{\partial\bar p}{\partial x}+F_0+\mathbb{E}[Y_iY_j]F_{ij}+f_{x,det},\equiv\mathbb{E}[\mathcal{L}]\\
		\dfrac{\diff Y_i}{\diff t}&=\left\langle\begin{bmatrix}
			f_{x,stoch}+Y_k(-\dfrac{\partial p_k}{\partial x}+F_k)+(Y_kY_l-\mathbb{E}[Y_kY_l])F_{kl}\\
			\vdots
		\end{bmatrix},\begin{bmatrix}
		u_i\\v_i\\w_i
		\end{bmatrix}\right\rangle,\\
		\dfrac{\partial u_i}{\partial t}&=\left[\mathbb{E}[\mathcal{L}Y_j]-\langle\mathbb{E}[\mathcal{L}Y_j],u_k\rangle u_k\right]C_{ij}^{-1}\\
		&=\left[-\dfrac{\partial p_k}{\partial x}\mathbb{E}[Y_kY_j]+F_k\mathbb{E}[Y_kY_j]+F_{kl}\mathbb{E}[Y_kY_lY_j]+\mathbb{E}[f_{x,stoch}Y_j]-\left\langle-\dfrac{\partial p_k}{\partial x}\mathbb{E}[Y_kY_j]+F_k\mathbb{E}[Y_kY_j]+F_{kl}\mathbb{E}[Y_kY_lY_j]+\mathbb{E}[f_{x,stoch}Y_j],u_k\right\rangle u_k\right]C_{ij}^{-1},
	\end{align*}
	with \begin{align*}
		C_{ij}^{-1}=\begin{cases}
			\delta_{ij}\mu_i,\quad\mu_i>\varepsilon\mu_{max},\\
			0,\quad\text{otherwise}.
		\end{cases}
	\end{align*}

	With this pseudoinverse:
	\paragraph{Modes}
	
	RHS:\begin{align*}
		\left(\left(-\dfrac{\partial p_k}{\partial x}+F_k\right)C_{ki}+F_{kl}M_{kli}+\mathbb{E}[f_{x,stoch}Y_i]-\left\langle(\dots),u_k\right\rangle u_k\right) / \mu_i,
	\end{align*}
	where $\mu_i$ is the $i$-th eigenvalue of the second moment $C_{ij}$ and $M_{kli}$ is the third moment.

	Explicit:\begin{align*}
		N&=\left\lbrace-\left(\bar u\dfrac{\partial u_k}{\partial x}+u_k\dfrac{\partial\bar u}{\partial x}+\bar v\dfrac{\partial u_k}{\partial y}+v_k\dfrac{\partial\bar u}{\partial y}+\bar w\dfrac{\partial u_k}{\partial z}+w_k\dfrac{\partial\bar u}{\partial z}\right)C_{ki}+F_{ml}M_{mli}+\mathbb{E}[f_{x,stoch}Y_i]-\left\langle-(\dots)+\nu\Delta u_n\mathbb{E}[Y_nY_i],u_p\right\rangle u_p\right\rbrace\cdot\dfrac{1}{\mu_i}\\
		&=-\left(\bar u\dfrac{\partial u_k}{\partial x}+u_k\dfrac{\partial\bar u}{\partial x}+\bar v\dfrac{\partial u_k}{\partial y}+v_k\dfrac{\partial\bar u}{\partial y}+\bar w\dfrac{\partial u_k}{\partial z}+w_k\dfrac{\partial\bar u}{\partial z}\right)+\dfrac{F_{ml}M_{mli}+\mathbb{E}[f_{x,stoch}Y_i]}{\mu_i}-\left\langle-(\dots)+\nu\Delta u_i,u_p\right\rangle u_p\\
		&=
		-\left(\bar u\frac{\partial u_i}{\partial x}+u_i\frac{\partial\bar u}{\partial x}
		+\bar v\frac{\partial u_i}{\partial y}+v_i\frac{\partial\bar u}{\partial y}
		+\bar w\frac{\partial u_i}{\partial z}+w_i\frac{\partial\bar u}{\partial z}\right)
		\;+\;\frac{F_{ml}M_{mli}}{\mu_i}\;+\;\frac{\mathbb{E}[f_{x,\mathrm{stoch}}Y_i]}{\mu_i}
		\;\\
		&-\;\left\langle
		-\left(\bar u\frac{\partial u_i}{\partial x}+u_i\frac{\partial\bar u}{\partial x}
		+\bar v\frac{\partial u_i}{\partial y}+v_i\frac{\partial\bar u}{\partial y}
		+\bar w\frac{\partial u_i}{\partial z}+w_i\frac{\partial\bar u}{\partial z}\right)
		+\nu\Delta u_i
		+\frac{F_{ml}M_{mli}+\mathbb{E}[f_{x,\mathrm{stoch}}Y_i]}{\mu_i}
		\;,\;u_p\right\rangle u_p.
	\end{align*}
	In these two above equations, the division over $\mu_i$ does NOT imply implicit summation. The rest are indeed dummy, summed-over, indices.
	
	Implicit:\begin{align*}
		\nu\Delta u_k\mathbb{E}[Y_kY_i] / \mu_i
	\end{align*}
	
	\paragraph{Mean} RHS:\begin{align*}
		-\dfrac{\partial\bar p}{\partial x}+F_0+C_{ij}F_{ij}+f_{x,det}.
	\end{align*}
	
	Explicit:\begin{align*}
		N=-\bar u\dfrac{\partial\bar u}{\partial x}-\bar v\dfrac{\partial\bar u}{\partial y}-\bar w\dfrac{\partial\bar u}{\partial z}-\left(u_i\dfrac{\partial u_j}{\partial x}+v_i\dfrac{\partial u_j}{\partial y}+w_i\dfrac{\partial u_j}{\partial z}\right)C_{ij}+f_{x,det}.
	\end{align*}
	
	Implicit:\begin{align*}
		\nu\Delta\bar u.
	\end{align*}

	Then, Poisson explicit problem is given by
	\begin{align*}
		\Delta p^{(n+1)}=\nabla\cdot\left(\dfrac{2\mathbf{u}^{(n)}-\frac{1}{2}\mathbf{u}^{(n-1)}}{\Delta t}+2\mathbf{N}^{(n)}-\mathbf{N}^{(n-1)}\right)=\dfrac{\nabla\cdot\mathbf{\hat u}^{(n)}}{\Delta t},
	\end{align*}
	where $\mathbf{\hat u}^{(n)}=2\mathbf{u}^{(n)}-\frac{1}{2}\mathbf{u}^{(n-1)}+\Delta t(2\mathbf{N}^{(n)}-\mathbf{N}^{(n-1)})$.
	
	Helmholtz implicit solve given component-wise by\begin{align*}
		\left(\Delta-\dfrac{3}{2\nu\Delta t}\right)u_k^{(n+1)}&=-\dfrac{1}{\nu\Delta t}\hat u_k^{(n)}+\dfrac{1}{\nu}\partial_k p^{(n+1)}
	\end{align*}
